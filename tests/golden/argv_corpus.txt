# argv corpus — the behavioural pin for the parser.
#
# Every line is one invocation and the outcome it must produce. This table is
# the parser's behavioural contract: a change to the parser is correct only
# when it still reproduces this table, and every disagreement is a decision
# someone has to make on purpose rather than discover in the field.
#
# Format:  argv | kind | command | actions | rc | verdict
#
#   kind     what the parse layer concluded:
#              ok            an action is dispatchable
#              help          usage requested
#              version_only  no action given
#              parse_error   the parser itself rejected the input
#              guard_error   a surface guard rejected it after a clean parse
#   command  the action selected, or - when none
#   actions  action-flag count feeding the mutual-exclusion verdict
#   rc       the exit code the parse layer settles on its own, measured against
#            a real binary. "--" wherever the answer is not the parse layer's:
#            for a dispatchable command the handler owns the outcome, so the
#            code there says nothing about parsing.
#   verdict  intended  the behaviour is wanted and must be kept
#            tolerated the behaviour is an accident of the current parser;
#                      it may change, but only deliberately

# --- Unknown input -----------------------------------------------------------
# A flag the parser does not know is reported by the parser; it is never
# absorbed into the install positional.
--definitely-not-a-real-flag   | parse_error  | -              | 0 | 1  | intended
--commit_update --bogus        | parse_error  | -              | 0 | 1  | intended
-ab                            | parse_error  | -              | 0 | 1  | intended

# --- Divergence surface vs. a getopt-style parser -----------------------------
# Abbreviations are rejected on purpose: exact_long_match() checks every
# long-option match, including ones getopt_long itself would resolve
# unambiguously.
--install_up x                 | parse_error  | -              | 0 | 1  | intended
# "--flag=value" is accepted for value-taking flags, the GNU convention.
--cancel_install=5             | ok           | cancel_install | 1 | -- | intended
# The "=value" acceptance above does not extend to switch/modifier flags —
# they are declared no_argument in the option table, so getopt_long itself
# rejects a "=value" suffix on them. No extra code; this falls out of the
# table for free.
--debug=x                      | parse_error  | -              | 0 | 1  | intended
--install_update=/path         | parse_error  | -              | 0 | 1  | intended
# The empty-value rule applies to the "=value" form too, not just
# "--flag value".
--cancel_install=              | parse_error  | -              | 0 | 1  | intended
--is_app_state_bad=A           | ok           | is_app_state_bad | 1 | -- | intended
# Operand-before-option ordering is accepted here; getopt_long permutes unless
# POSIXLY_CORRECT is set, but this parser's optstring leading '-' returns
# operands in argv order and is immune to the environment either way —
# pinned by Operands.OrderingImmuneToPosixlyCorrect.
x --install_update             | ok           | install_update | 1 | -- | tolerated
# Repeating a modifier is accepted; only actions and help reject a repeat.
--debug --debug                | version_only | -              | 0 | 0  | tolerated
# A required-value option consumes the very next token unconditionally, even
# one that looks like another flag — getopt_long does not special-case this.
# The value then fails parse_u32 on its own merits (not digits-only).
--cancel_install --debug       | parse_error  | -              | 0 | 1  | intended
# A second operand has nowhere to go — the bare-path rule already limits the install
# path to one positional; a second one is rejected outright rather than
# silently discarded.
x y                            | parse_error  | -              | 0 | 1  | intended

# --- Value handling ----------------------------------------------------------
# An empty value is rejected rather than replaced by the argument's default.
--is_app_state_bad ""          | parse_error  | -              | 0 | 1  | intended
--cancel_install ""            | parse_error  | -              | 0 | 1  | intended
--cancel_install               | parse_error  | -              | 0 | 1  | intended
--is_app_state_bad             | parse_error  | -              | 0 | 1  | intended
# Type conversion is the standard library's, not the parser's — a hand-written
# port inherits none of it and must reimplement each of these on purpose.
--cancel_install abc           | parse_error  | -              | 0 | 1  | intended
--cancel_install 5x            | parse_error  | -              | 0 | 1  | intended
# Decimal digits only, over the full token: "0x10" is rejected.
--cancel_install 0x10          | parse_error  | -              | 0 | 1  | intended
--cancel_install 99999999999   | parse_error  | -              | 0 | 1  | intended
# A leading '-' is rejected outright: an unsigned extraction that wraps "-1"
# to 4294967295 would dispatch a real cancel call.
--cancel_install -1            | parse_error  | -              | 0 | 1  | intended
--is_app_state_bad toolong     | parse_error  | -              | 0 | 1  | intended
--is_app_state_bad Z           | ok           | is_app_state_bad | 1 | -- | tolerated
#   ^ no constraint is attached; "accepted states: A or B" is help text only, so
#     the value reaches the handler and is rejected there.
--is_app_state_bad A           | ok           | is_app_state_bad | 1 | -- | intended

# --- Ordering: errors, help and repeats --------------------------------------
# --help does not end the scan: the rest of argv is still read, and a later
# error wins.
--help --bogus                 | parse_error  | -                | 0 | 1  | intended
# An error ahead of --help is reached before help is.
--bogus --help                 | parse_error  | -                | 0 | 1  | intended
--help --cancel_install        | parse_error  | -                | 0 | 1  | intended
# Bundled short option: 'h' is recorded and 'x' on the same token is still seen.
-hx                            | parse_error  | -                | 0 | 1  | intended
# An extra operand alongside --help is ignored: help is honoured before
# classify(), which would reject a bare positional via the bare-path rule, runs.
--help x                       | help         | -                | 0 | 0  | intended
# A lone "-" is rejected as an unrecognized argument, ahead of a literal "--".
-                              | parse_error  | -                | 0 | 1  | intended
# The same "-" is a legitimate operand once "--" has ended option parsing.
--install_update -- -          | ok           | install_update   | 1 | -- | intended
# Repeating a switch action is rejected.
--commit_update --commit_update | parse_error | -                | 0 | 1  | intended
# Repeating a value action is rejected: keeping only the last value would swap
# the session of --cancel_install without a word.
--cancel_install 5 --cancel_install 6 | parse_error | -           | 0 | 1  | intended

# --- Repeats, aliases and value edge cases -----------------------------------
# Repeating help is rejected like repeating an action; modifiers stay lenient.
--help --help                  | parse_error  | -                | 0 | 1  | intended
-h -h                          | parse_error  | -                | 0 | 1  | intended
# --ignore_rest is not an alias of "--": "--" is the only form.
--ignore_rest                  | parse_error  | -                | 0 | 1  | intended
# State values: the parser checks one-character syntax only and the command
# checks the value, so a space reaches the command (which rejects it) while a
# space-plus-letter is two characters and fails here.
--set_app_state_bad <SP>       | ok           | set_app_state_bad | 1 | -- | intended
--is_app_state_bad <SP>A       | parse_error  | -                | 0 | 1  | intended
# An extra operand after "--" is refused: the single-operand limit applies
# there as it does without "--".
--install_update -- x y        | parse_error  | -                | 0 | 1  | intended
# An ambiguous prefix is refused for the same reason a resolvable one is: the
# parser never guesses which flag was meant.
--install                      | parse_error  | -                | 0 | 1  | intended
# A shortened form of a full name is rejected even where getopt_long could
# resolve it unambiguously: the exact name is required.
--update_reboot_stat           | parse_error  | -                | 0 | 1  | intended
# Precedence: a parse error beats help (pinned above), and help beats every
# later verdict — the mutual-exclusion count and the install-surface guards.
--help --apply_update --commit_update | help   | -                | 0 | 0  | intended
--help --detach                | help         | -                | 0 | 0  | intended

# --- Dispatch counting -------------------------------------------------------
--serial                       | version_only | -              | 0 | 0  | intended
--debug --serial               | version_only | -              | 0 | 0  | intended
--apply_update --commit_update | combo_error  | -              | 2 | 65 | intended
#   ^ two actions: rejected by the mutual-exclusion verdict, no handler runs.

# --- Install surface guards --------------------------------------------------
x                              | guard_error  | -              | 0 | 65 | intended
--detach                       | guard_error  | -              | 0 | 65 | intended
--install_update ""            | guard_error  | -              | 0 | 61 | intended
# "--" is the POSIX end of options: getopt_long returns -1 at it instead of
# yielding further operands, so what follows is collected in a dedicated pass,
# under the same single-operand limit. The bare-path rule still applies: a path with no
# --install_update is rejected.
-- x                           | guard_error  | -              | 0 | 65 | intended
--install_update -- x          | ok           | install_update | 1 | -- | intended
